const API_BASE_URL = (process.env.NEXT_PUBLIC_API_URL || 'http://localhost:5000').replace(/\/+$/, '');

async function request(path, body, method = 'PATCH') {
  const response = await fetch(`${API_BASE_URL}/api/merchant/orders${path}`, {
    method, credentials: 'include', headers: { 'Content-Type': 'application/json' }, ...(method === 'PATCH' ? { body: JSON.stringify(body) } : {}),
  });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload.message || 'Order request failed');
  return payload;
}

export async function getMerchantOrders() {
  const response = await fetch(`${API_BASE_URL}/api/merchant/orders`, { credentials: 'include' });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload.message || 'Failed to load orders');
  return payload.map((order) => {
    const group = order.merchantGroups?.[0] || {};
    const status = group.status || order.status || 'pending';
    return {
      id: order._id,
      orderType: 'direct',
      orderStatus: status,
      createdAt: order.createdAt,
      updatedAt: order.updatedAt,
      customer: { name: order.userInfo?.name || 'Customer', email: order.userInfo?.email || '', phone: order.userInfo?.phone || '' },
      items: (order.products || []).map((line) => ({ name: line.name, quantity: line.quantity, price: line.price, image: line.image })),
      total: (order.products || []).reduce((sum, line) => sum + line.price * line.quantity, 0),
      subtotal: (order.products || []).reduce((sum, line) => sum + line.price * line.quantity, 0),
      deliveryFee: 0,
      shippingAddress: { town: '', county: '' },
      paymentStatus: group.paymentConfirmedAt ? 'confirmed' : group.paymentSentAt ? 'sent' : 'not recorded',
      rejectionReason: group.rejectionReason || null,
      paymentInstruction: group.paymentInstruction || '',
      paymentLink: group.paymentLink || '',
      paymentSentAt: group.paymentSentAt || null,
      paymentReceivedAt: group.paymentReceivedAt || null,
    };
  });
}

export const decideMerchantOrder = (id, decision, reason) => request(`/${encodeURIComponent(id)}/decision`, { decision, reason });
export const progressMerchantOrder = (id, status) => request(`/${encodeURIComponent(id)}/status`, { status });
export const savePaymentInstructions = (id, instruction, link) => request(`/${encodeURIComponent(id)}/payment-instructions`, { instruction, link });
export const confirmPaymentReceived = (id) => request(`/${encodeURIComponent(id)}/payment-received`, {}, 'PATCH');

"use client";

import {
  createContext,
  useCallback,
  useEffect,
  useContext,
  useMemo,
  useState,
} from "react";
import { confirmPaymentReceived, decideMerchantOrder, getMerchantOrders, progressMerchantOrder, savePaymentInstructions } from "../Services/orderService";

const OrderContext = createContext(null);

export function OrderProvider({ children }) {
  const [orders, setOrders] = useState([]);
  const [loadError, setLoadError] = useState('');
  const refreshOrders = useCallback(async () => {
    try { setOrders(await getMerchantOrders()); setLoadError(''); return true; }
    catch (error) { setLoadError(error.message); return false; }
  }, []);
  useEffect(() => {
    let active = true;
    getMerchantOrders().then((rows) => { if (active) setOrders(rows); })
      .catch((error) => { if (active) setLoadError(error.message); });
    return () => { active = false; };
  }, []);

  // Filter orders by type or fulfillment status.
  const getOrders = useCallback(
    ({ orderType = "all", orderStatus = "all" } = {}) => {
      return orders.filter((order) => {
        const matchesType =
          orderType === "all" || order.orderType === orderType;

        const matchesStatus =
          orderStatus === "all" || order.orderStatus === orderStatus;

        return matchesType && matchesStatus;
      });
    },
    [orders]
  );

  // Update one order without replacing the entire order list.
  const updateOrder = useCallback((orderId, updates) => {
    setOrders((currentOrders) =>
      currentOrders.map((order) =>
        order.id === orderId
          ? {
              ...order,
              ...updates,
              updatedAt: new Date().toISOString(),
            }
          : order
      )
    );
  }, []);

  // Only pending orders can be accepted.
  const acceptOrder = useCallback(
    async (orderId) => {
      const order = orders.find((item) => item.id === orderId);

      if (!order || order.orderStatus !== "pending") {
        return false;
      }

      try { await decideMerchantOrder(orderId, 'accept'); } catch { return false; }
      updateOrder(orderId, { orderStatus: "accepted", rejectionReason: null });

      return true;
    },
    [orders, updateOrder]
  );

  // Only pending orders can be rejected.
  const rejectOrder = useCallback(
    async (orderId, reason = "Order rejected by merchant.") => {
      const order = orders.find((item) => item.id === orderId);

      if (!order || order.orderStatus !== "pending") {
        return false;
      }

      try { await decideMerchantOrder(orderId, 'reject', reason); } catch { return false; }
      updateOrder(orderId, {
        orderStatus: "rejected",
        rejectionReason: reason,
      });

      return true;
    },
    [orders, updateOrder]
  );

  // Enforce the normal fulfillment sequence.
  const advanceOrder = useCallback(
    async (orderId, nextStatus) => {
      const allowedTransitions = {
        accepted: ["processing"],
        processing: ["dispatched"],
        dispatched: ["delivered"],
      };

      const order = orders.find((item) => item.id === orderId);

      if (
        !order ||
        !allowedTransitions[order.orderStatus]?.includes(nextStatus)
      ) {
        return false;
      }

      try { await progressMerchantOrder(orderId, nextStatus); } catch { return false; }
      updateOrder(orderId, { orderStatus: nextStatus });

      return true;
    },
    [orders, updateOrder]
  );

  // Record receipt only after an order has been delivered.
  const confirmCustomerReceipt = useCallback(
    (orderId) => {
      const order = orders.find((item) => item.id === orderId);

      if (
        !order ||
        order.orderStatus !== "delivered" ||
        order.customerReceiptStatus === "confirmed"
      ) {
        return false;
      }

      updateOrder(orderId, {
        customerReceiptStatus: "confirmed",
      });

      return true;
    },
    [orders, updateOrder]
  );

  const updatePaymentInstructions = useCallback(async (orderId, instruction, link) => {
    try {
      const saved = await savePaymentInstructions(orderId, instruction, link);
      const group = saved.merchantGroup;
      updateOrder(orderId, { paymentInstruction: group?.paymentInstruction || '', paymentLink: group?.paymentLink || '' });
      return true;
    } catch { return false; }
  }, [updateOrder]);

  const markPaymentReceived = useCallback(async (orderId) => {
    try { await confirmPaymentReceived(orderId); updateOrder(orderId, { paymentReceivedAt: new Date().toISOString() }); return true; }
    catch { return false; }
  }, [updateOrder]);

  // Summary counts for dashboard cards.
  const stats = useMemo(
    () => ({
      total: orders.length,

      direct: orders.filter(
        (order) => order.orderType === "direct"
      ).length,

      affiliate: orders.filter(
        (order) => order.orderType === "affiliate"
      ).length,

      pending: orders.filter(
        (order) => order.orderStatus === "pending"
      ).length,

      processing: orders.filter((order) =>
        ["accepted", "processing", "ready"].includes(order.orderStatus)
      ).length,

      dispatched: orders.filter(
        (order) => order.orderStatus === "dispatched"
      ).length,

      delivered: orders.filter(
        (order) => order.orderStatus === "delivered"
      ).length,

      rejected: orders.filter(
        (order) => order.orderStatus === "rejected"
      ).length,
    }),
    [orders]
  );

  const value = useMemo(
    () => ({
      orders,
      loadError,
      refreshOrders,
      stats,
      getOrders,
      updateOrder,
      acceptOrder,
      rejectOrder,
      advanceOrder,
      confirmCustomerReceipt,
      updatePaymentInstructions,
      markPaymentReceived,
    }),
    [
      orders,
      loadError,
      refreshOrders,
      stats,
      getOrders,
      updateOrder,
      acceptOrder,
      rejectOrder,
      advanceOrder,
      confirmCustomerReceipt,
      updatePaymentInstructions,
      markPaymentReceived,
    ]
  );

  return (
    <OrderContext.Provider value={value}>
      {children}
    </OrderContext.Provider>
  );
}

export function useOrders() {
  const context = useContext(OrderContext);

  if (!context) {
    throw new Error("useOrders must be used inside an OrderProvider");
  }

  return context;
}

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,

import { usePathname, useRouter } from "next/navigation";

const MerchantAuthContext = createContext(null);
const publicPaths = new Set(["/login", "/register"]);

function AuthLoading({ message = "Checking your merchant session…" }) {
  return (
    <main className="flex min-h-screen items-center justify-center bg-slate-50 px-4 text-slate-700">
      <div role="status" className="flex items-center gap-3 rounded-xl border border-slate-200 bg-white px-5 py-4 text-sm shadow-sm">
        <span aria-hidden="true" className="h-4 w-4 animate-spin rounded-full border-2 border-emerald-600 border-t-transparent" />

      </div>
    </main>

function MerchantRouteBoundary({ children }) {
  const pathname = usePathname() || "/";
  const router = useRouter();
  const { status, restoreSession } = useMerchantAuth();
  const isPublic = publicPaths.has(pathname);

  useEffect(() => {
    if (status === "unauthenticated" && !isPublic) {
      router.replace(`/login?next=${encodeURIComponent(pathname)}`);
    } else if (status === "authenticated" && isPublic) {
      router.replace("/");
    }
  }, [isPublic, pathname, router, status]);

  if (status === "loading") return <AuthLoading />;

  if (status === "authenticated" && isPublic) return <AuthLoading message="Opening your merchant portal…" />;
  if (status === "unauthenticated" && !isPublic) return <AuthLoading message="Taking you to sign in…" />;

  if (status === "error" && !isPublic) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-slate-50 px-4">
        <section className="w-full max-w-md rounded-2xl border border-slate-200 bg-white p-6 text-center shadow-sm">
          <h1 className="text-lg font-semibold text-slate-900">We couldn’t verify your session</h1>
          <p className="mt-2 text-sm leading-6 text-slate-600">The authentication service could not be reached. Try again, or sign in once the service is available.</p>
          <div className="mt-5 flex flex-wrap justify-center gap-3">
            <button type="button" onClick={restoreSession} className="rounded-lg bg-emerald-700 px-4 py-2.5 text-sm font-semibold text-white hover:bg-emerald-800">Try again</button>
            <a href="/login" className="rounded-lg border border-slate-300 px-4 py-2.5 text-sm font-semibold text-slate-700 hover:bg-slate-50">Sign in</a>
          </div>
        </section>
      </main>
    );
  }

  return children;
}

export function MerchantAuthProvider({ children }) {
  const [merchant, setMerchant] = useState(null);
  const [status, setStatus] = useState("loading");
  const [logoutError, setLogoutError] = useState("");
  const [loggingOut, setLoggingOut] = useState(false);

  const restoreSession = useCallback(async () => {
    setStatus("loading");
    try {
      const currentMerchant = await merchantAuthService.getCurrentMerchant();
      setMerchant(currentMerchant);
      setStatus("authenticated");
      return currentMerchant;
    } catch (error) {
      setMerchant(null);
      if (isUnauthenticatedError(error)) {
        setStatus("unauthenticated");
      } else {
        setStatus("error");
      }
      return null;
    }
  }, []);

  useEffect(() => {
    let active = true;

    merchantAuthService.getCurrentMerchant().then((currentMerchant) => {
      if (!active) return;
      setMerchant(currentMerchant);
      setStatus("authenticated");
    }).catch((error) => {
      if (!active) return;
      setMerchant(null);
      setStatus(isUnauthenticatedError(error) ? "unauthenticated" : "error");
    });

    return () => {
      active = false;
    };
  }, []);

  const acceptAuthenticatedMerchant = useCallback((profile) => {
    setMerchant(profile);
    setLogoutError("");
    setStatus("authenticated");
  }, []);

  const logout = useCallback(async () => {
    setLoggingOut(true);
    setLogoutError("");
    try {
      await merchantAuthService.logout();
      setMerchant(null);
      setStatus("unauthenticated");
      return { success: true };
    } catch (error) {
      const message = error.message || "Could not sign out. Please try again.";
      setLogoutError(message);
      return { success: false, error: message };
    } finally {
      setLoggingOut(false);
    }
  }, []);

  const value = useMemo(() => ({
    merchant,
    status,
    logoutError,
    loggingOut,
    restoreSession,
    acceptAuthenticatedMerchant,
    logout,
  }), [merchant, status, logoutError, loggingOut, restoreSession, acceptAuthenticatedMerchant, logout]);

  return (
    <MerchantAuthContext.Provider value={value}>
      <MerchantRouteBoundary>{children}</MerchantRouteBoundary>
    </MerchantAuthContext.Provider>
  );
}

export function useMerchantAuth() {
  const context = useContext(MerchantAuthContext);
  if (!context) throw new Error("useMerchantAuth must be used inside MerchantAuthProvider.");
  return context;
}

const API_BASE_URL = (
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000"
).replace(/\/+$/, "");

export class MerchantAuthError extends Error {
  constructor(message, status = 0, body = {}) {
    super(message);
    this.name = "MerchantAuthError";
    this.status = status;
    this.body = body;
  }
}

async function request(path, { method = "GET", body, signal } = {}) {
  let response;

  try {
    response = await fetch(`${API_BASE_URL}/api/merchant-auth${path}`, {
      method,
      credentials: "include",
      cache: "no-store",
      headers: body ? { "Content-Type": "application/json" } : undefined,
      body: body ? JSON.stringify(body) : undefined,
      signal: signal || AbortSignal.timeout(10000),
    });
  } catch (error) {
    const message = error?.name === "TimeoutError"
      ? "The authentication service took too long to respond. Please try again."
      : "Could not reach the authentication service. Check that it is running and try again.";
    throw new MerchantAuthError(message);
  }

  let payload = {};
  try {
    const text = await response.text();
    payload = text ? JSON.parse(text) : {};
  } catch {
    // Non-JSON or malformed responses are handled using status-based messages.
  }

  if (!response.ok) {
    let message;
    if (response.status === 401) {
      message = "Invalid email/phone or password.";
    } else if (response.status === 409) {
      message = "An account with this email or phone number already exists.";
    } else if (response.status === 400) {
      message = payload.message && payload.message !== "Invalid request"
        ? payload.message
        : "Check the information entered and try again.";
    } else if (response.status >= 500) {
      message = "The authentication service is temporarily unavailable. Please try again.";
    } else if (payload.message && response.status < 500) {
      message = payload.message;
    } else {
      message = `The request could not be completed (HTTP ${response.status}).`;
    }
    throw new MerchantAuthError(message, response.status, payload);
  }

  return payload;
}

function safeMerchant(payload) {
  const merchant = payload?.merchant;
  if (!merchant || typeof merchant !== "object") {
    throw new MerchantAuthError("The server returned an invalid merchant profile.");
  }

  // Allow only the safe profile fields returned by the merchant auth API.
  return {
    id: typeof merchant.id === "string" ? merchant.id : "",
    name: typeof merchant.name === "string" ? merchant.name : "",
    email: typeof merchant.email === "string" ? merchant.email : "",
    phone: typeof merchant.phone === "string" ? merchant.phone : "",
    createdAt: typeof merchant.createdAt === "string" ? merchant.createdAt : null,
    updatedAt: typeof merchant.updatedAt === "string" ? merchant.updatedAt : null,
  };
}

export const merchantAuthService = {
  async register({ name, email, phone, password }) {
    const payload = await request("/register", {
      method: "POST",
      body: { name, email, phone, password },
    });
    return safeMerchant(payload);
  },

  async login({ identifier, password }) {
    const payload = await request("/login", {
      method: "POST",
      body: { identifier, password },
    });
    return safeMerchant(payload);
  },

  async getCurrentMerchant() {
    const payload = await request("/me");
    return safeMerchant(payload);
  },

  async logout() {
    await request("/logout", { method: "POST", body: {} });
  },
};

export function isUnauthenticatedError(error) {
  return error instanceof MerchantAuthError && error.status === 401;
}

const API_BASE_URL = (process.env.NEXT_PUBLIC_API_URL || 'http://localhost:5000').replace(/\/+$/, '');

async function request(path, body, method = 'PATCH') {
  const response = await fetch(`${API_BASE_URL}/api/merchant/orders${path}`, {
    method, credentials: 'include', headers: { 'Content-Type': 'application/json' }, ...(method === 'PATCH' ? { body: JSON.stringify(body) } : {}),
  });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload.message || 'Order request failed');
  return payload;
}

export async function getMerchantOrders() {
  const response = await fetch(`${API_BASE_URL}/api/merchant/orders`, { credentials: 'include' });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload.message || 'Failed to load orders');
  return payload.map((order) => {
    const group = order.merchantGroups?.[0] || {};
    const status = group.status || order.status || 'pending';
    return {
      id: order._id,
      orderType: 'direct',
      orderStatus: status,
      createdAt: order.createdAt,
      updatedAt: order.updatedAt,
      customer: { name: order.userInfo?.name || 'Customer', email: order.userInfo?.email || '', phone: order.userInfo?.phone || '' },
      items: (order.products || []).map((line) => ({ name: line.name, quantity: line.quantity, price: line.price, image: line.image })),
      total: (order.products || []).reduce((sum, line) => sum + line.price * line.quantity, 0),
      subtotal: (order.products || []).reduce((sum, line) => sum + line.price * line.quantity, 0),
      deliveryFee: 0,
      shippingAddress: { town: '', county: '' },
      paymentStatus: group.paymentConfirmedAt ? 'confirmed' : group.paymentSentAt ? 'sent' : 'not recorded',
      rejectionReason: group.rejectionReason || null,
      paymentInstruction: group.paymentInstruction || '',
      paymentLink: group.paymentLink || '',
      paymentSentAt: group.paymentSentAt || null,
      paymentReceivedAt: group.paymentReceivedAt || null,
    };
  });
}

export const decideMerchantOrder = (id, decision, reason) => request(`/${encodeURIComponent(id)}/decision`, { decision, reason });
export const progressMerchantOrder = (id, status) => request(`/${encodeURIComponent(id)}/status`, { status });
export const savePaymentInstructions = (id, instruction, link) => request(`/${encodeURIComponent(id)}/payment-instructions`, { instruction, link });
export const confirmPaymentReceived = (id) => request(`/${encodeURIComponent(id)}/payment-received`, {}, 'PATCH');

"use client";

import {
  createContext,
  useCallback,
  useEffect,
  useContext,
  useMemo,
  useState,
} from "react";
import { confirmPaymentReceived, decideMerchantOrder, getMerchantOrders, progressMerchantOrder, savePaymentInstructions } from "../Services/orderService";

const OrderContext = createContext(null);

export function OrderProvider({ children }) {
  const [orders, setOrders] = useState([]);
  const [loadError, setLoadError] = useState('');
  const refreshOrders = useCallback(async () => {
    try { setOrders(await getMerchantOrders()); setLoadError(''); return true; }
    catch (error) { setLoadError(error.message); return false; }
  }, []);
  useEffect(() => {
    let active = true;
    getMerchantOrders().then((rows) => { if (active) setOrders(rows); })
      .catch((error) => { if (active) setLoadError(error.message); });
    return () => { active = false; };
  }, []);

  // Filter orders by type or fulfillment status.
  const getOrders = useCallback(
    ({ orderType = "all", orderStatus = "all" } = {}) => {
      return orders.filter((order) => {
        const matchesType =
          orderType === "all" || order.orderType === orderType;

        const matchesStatus =
          orderStatus === "all" || order.orderStatus === orderStatus;

        return matchesType && matchesStatus;
      });
    },
    [orders]
  );

  // Update one order without replacing the entire order list.
  const updateOrder = useCallback((orderId, updates) => {
    setOrders((currentOrders) =>
      currentOrders.map((order) =>
        order.id === orderId
          ? {
              ...order,
              ...updates,
              updatedAt: new Date().toISOString(),
            }
          : order
      )
    );
  }, []);

  // Only pending orders can be accepted.
  const acceptOrder = useCallback(
    async (orderId) => {
      const order = orders.find((item) => item.id === orderId);

      if (!order || order.orderStatus !== "pending") {
        return false;
      }

      try { await decideMerchantOrder(orderId, 'accept'); } catch { return false; }
      updateOrder(orderId, { orderStatus: "accepted", rejectionReason: null });

      return true;
    },
    [orders, updateOrder]
  );

  // Only pending orders can be rejected.
  const rejectOrder = useCallback(
    async (orderId, reason = "Order rejected by merchant.") => {
      const order = orders.find((item) => item.id === orderId);

      if (!order || order.orderStatus !== "pending") {
        return false;
      }

      try { await decideMerchantOrder(orderId, 'reject', reason); } catch { return false; }
      updateOrder(orderId, {
        orderStatus: "rejected",
        rejectionReason: reason,
      });

      return true;
    },
    [orders, updateOrder]
  );

  // Enforce the normal fulfillment sequence.
  const advanceOrder = useCallback(
    async (orderId, nextStatus) => {
      const allowedTransitions = {
        accepted: ["processing"],
        processing: ["dispatched"],
        dispatched: ["delivered"],
      };

      const order = orders.find((item) => item.id === orderId);

      if (
        !order ||
        !allowedTransitions[order.orderStatus]?.includes(nextStatus)
      ) {
        return false;
      }

      try { await progressMerchantOrder(orderId, nextStatus); } catch { return false; }
      updateOrder(orderId, { orderStatus: nextStatus });

      return true;
    },
    [orders, updateOrder]
  );

  // Record receipt only after an order has been delivered.
  const confirmCustomerReceipt = useCallback(
    (orderId) => {
      const order = orders.find((item) => item.id === orderId);

      if (
        !order ||
        order.orderStatus !== "delivered" ||
        order.customerReceiptStatus === "confirmed"
      ) {
        return false;
      }

      updateOrder(orderId, {
        customerReceiptStatus: "confirmed",
      });

      return true;
    },
    [orders, updateOrder]
  );

  const updatePaymentInstructions = useCallback(async (orderId, instruction, link) => {
    try {
      const saved = await savePaymentInstructions(orderId, instruction, link);
      const group = saved.merchantGroup;
      updateOrder(orderId, { paymentInstruction: group?.paymentInstruction || '', paymentLink: group?.paymentLink || '' });
      return true;
    } catch { return false; }
  }, [updateOrder]);

  const markPaymentReceived = useCallback(async (orderId) => {
    try { await confirmPaymentReceived(orderId); updateOrder(orderId, { paymentReceivedAt: new Date().toISOString() }); return true; }
    catch { return false; }
  }, [updateOrder]);

  // Summary counts for dashboard cards.
  const stats = useMemo(
    () => ({
      total: orders.length,

      direct: orders.filter(
        (order) => order.orderType === "direct"
      ).length,

      affiliate: orders.filter(
        (order) => order.orderType === "affiliate"
      ).length,

      pending: orders.filter(
        (order) => order.orderStatus === "pending"
      ).length,

      processing: orders.filter((order) =>
        ["accepted", "processing", "ready"].includes(order.orderStatus)
      ).length,

      dispatched: orders.filter(
        (order) => order.orderStatus === "dispatched"
      ).length,

      delivered: orders.filter(
        (order) => order.orderStatus === "delivered"
      ).length,

      rejected: orders.filter(
        (order) => order.orderStatus === "rejected"
      ).length,
    }),
    [orders]
  );

  const value = useMemo(
    () => ({
      orders,
      loadError,
      refreshOrders,
      stats,
      getOrders,
      updateOrder,
      acceptOrder,
      rejectOrder,
      advanceOrder,
      confirmCustomerReceipt,
      updatePaymentInstructions,
      markPaymentReceived,
    }),
    [
      orders,
      loadError,
      refreshOrders,
      stats,
      getOrders,
      updateOrder,
      acceptOrder,
      rejectOrder,
      advanceOrder,
      confirmCustomerReceipt,
      updatePaymentInstructions,
      markPaymentReceived,
    ]
  );

  return (
    <OrderContext.Provider value={value}>
      {children}
    </OrderContext.Provider>
  );
}

export function useOrders() {
  const context = useContext(OrderContext);

  if (!context) {
    throw new Error("useOrders must be used inside an OrderProvider");
  }

  return context;
}

function AuthLoading({ message = "Checking your merchant session…" }) {
  return (

function MerchantRouteBoundary({ children }) {
  const pathname = usePathname() || "/";
  const router = useRouter();
  const { status, restoreSession } = useMerchantAuth();
  const isPublic = publicPaths.has(pathname);

  useEffect(() => {
    if (status === "unauthenticated" && !isPublic) {
      router.replace(`/login?next=${encodeURIComponent(pathname)}`);
    } else if (status === "authenticated" && isPublic) {
      router.replace("/");
    }
  }, [isPublic, pathname, router, status]);

  if (status === "loading") return <AuthLoading />;

  if (status === "authenticated" && isPublic) return <AuthLoading message="Opening your merchant portal…" />;
  if (status === "unauthenticated" && !isPublic) return <AuthLoading message="Taking you to sign in…" />;

  if (status === "error" && !isPublic) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-slate-50 px-4">
        <section className="w-full max-w-md rounded-2xl border border-slate-200 bg-white p-6 text-center shadow-sm">
          <h1 className="text-lg font-semibold text-slate-900">We couldn’t verify your session</h1>
          <p className="mt-2 text-sm leading-6 text-slate-600">The authentication service could not be reached. Try again, or sign in once the service is available.</p>
          <div className="mt-5 flex flex-wrap justify-center gap-3">
            <button type="button" onClick={restoreSession} className="rounded-lg bg-emerald-700 px-4 py-2.5 text-sm font-semibold text-white hover:bg-emerald-800">Try again</button>
            <a href="/login" className="rounded-lg border border-slate-300 px-4 py-2.5 text-sm font-semibold text-slate-700 hover:bg-slate-50">Sign in</a>
          </div>
        </section>
      </main>
    );
  }

  return children;
}

export function MerchantAuthProvider({ children }) {
  const [merchant, setMerchant] = useState(null);
  const [status, setStatus] = useState("loading");
  const [logoutError, setLogoutError] = useState("");
  const [loggingOut, setLoggingOut] = useState(false);

  const restoreSession = useCallback(async () => {
    setStatus("loading");
    try {
      const currentMerchant = await merchantAuthService.getCurrentMerchant();
      setMerchant(currentMerchant);
      setStatus("authenticated");
      return currentMerchant;
    } catch (error) {
      setMerchant(null);
      if (isUnauthenticatedError(error)) {
        setStatus("unauthenticated");
      } else {
        setStatus("error");
      }
      return null;
    }
  }, []);

  useEffect(() => {
    let active = true;

    merchantAuthService.getCurrentMerchant().then((currentMerchant) => {
      if (!active) return;
      setMerchant(currentMerchant);
      setStatus("authenticated");
    }).catch((error) => {
      if (!active) return;
      setMerchant(null);
      setStatus(isUnauthenticatedError(error) ? "unauthenticated" : "error");
    });

    return () => {
      active = false;
    };
  }, []);

  const acceptAuthenticatedMerchant = useCallback((profile) => {
    setMerchant(profile);
    setLogoutError("");
    setStatus("authenticated");
  }, []);

  const logout = useCallback(async () => {
    setLoggingOut(true);
    setLogoutError("");
    try {
      await merchantAuthService.logout();
      setMerchant(null);
      setStatus("unauthenticated");
      return { success: true };
    } catch (error) {
      const message = error.message || "Could not sign out. Please try again.";
      setLogoutError(message);
      return { success: false, error: message };
    } finally {
      setLoggingOut(false);
    }
  }, []);

  const value = useMemo(() => ({
    merchant,
    status,
    logoutError,
    loggingOut,
    restoreSession,
    acceptAuthenticatedMerchant,
    logout,
  }), [merchant, status, logoutError, loggingOut, restoreSession, acceptAuthenticatedMerchant, logout]);

  return (
    <MerchantAuthContext.Provider value={value}>
      <MerchantRouteBoundary>{children}</MerchantRouteBoundary>
    </MerchantAuthContext.Provider>
  );
}

export function useMerchantAuth() {
  const context = useContext(MerchantAuthContext);
  if (!context) throw new Error("useMerchantAuth must be used inside MerchantAuthProvider.");
  return context;
}

const API_BASE_URL = (
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000"
).replace(/\/+$/, "");

export class MerchantAuthError extends Error {
  constructor(message, status = 0, body = {}) {
    super(message);
    this.name = "MerchantAuthError";
    this.status = status;
    this.body = body;
  }
}

async function request(path, { method = "GET", body, signal } = {}) {
  let response;

  try {
    response = await fetch(`${API_BASE_URL}/api/merchant-auth${path}`, {
      method,
      credentials: "include",
      cache: "no-store",
      headers: body ? { "Content-Type": "application/json" } : undefined,
      body: body ? JSON.stringify(body) : undefined,
      signal: signal || AbortSignal.timeout(10000),
    });
  } catch (error) {
    const message = error?.name === "TimeoutError"
      ? "The authentication service took too long to respond. Please try again."
      : "Could not reach the authentication service. Check that it is running and try again.";
    throw new MerchantAuthError(message);
  }

  let payload = {};
  try {
    const text = await response.text();
    payload = text ? JSON.parse(text) : {};
  } catch {
    // Non-JSON or malformed responses are handled using status-based messages.
  }

  if (!response.ok) {
    let message;
    if (response.status === 401) {
      message = "Invalid email/phone or password.";
    } else if (response.status === 409) {
      message = "An account with this email or phone number already exists.";
    } else if (response.status === 400) {
      message = payload.message && payload.message !== "Invalid request"
        ? payload.message
        : "Check the information entered and try again.";
    } else if (response.status >= 500) {
      message = "The authentication service is temporarily unavailable. Please try again.";
    } else if (payload.message && response.status < 500) {
      message = payload.message;
    } else {
      message = `The request could not be completed (HTTP ${response.status}).`;
    }
    throw new MerchantAuthError(message, response.status, payload);
  }

  return payload;
}

function safeMerchant(payload) {
  const merchant = payload?.merchant;
  if (!merchant || typeof merchant !== "object") {
    throw new MerchantAuthError("The server returned an invalid merchant profile.");
  }

  // Allow only the safe profile fields returned by the merchant auth API.
  return {
    id: typeof merchant.id === "string" ? merchant.id : "",
    name: typeof merchant.name === "string" ? merchant.name : "",
    email: typeof merchant.email === "string" ? merchant.email : "",
    phone: typeof merchant.phone === "string" ? merchant.phone : "",
    createdAt: typeof merchant.createdAt === "string" ? merchant.createdAt : null,
    updatedAt: typeof merchant.updatedAt === "string" ? merchant.updatedAt : null,
  };
}

export const merchantAuthService = {
  async register({ name, email, phone, password }) {
    const payload = await request("/register", {
      method: "POST",
      body: { name, email, phone, password },
    });
    return safeMerchant(payload);
  },

  async login({ identifier, password }) {
    const payload = await request("/login", {
      method: "POST",
      body: { identifier, password },
    });
    return safeMerchant(payload);
  },

  async getCurrentMerchant() {
    const payload = await request("/me");
    return safeMerchant(payload);
  },

  async logout() {
    await request("/logout", { method: "POST", body: {} });
  },
};

export function isUnauthenticatedError(error) {
  return error instanceof MerchantAuthError && error.status === 401;
}
